### 3. `src/gpu.rs` — Add check_safety method
Add to `impl HardwareBridge` (before `apply_emergency_brake`):
```rust
pub fn check_safety(telemetry: &GpuTelemetry) -> SafetyStatus {
    // In software-only mode (simulated values), skip checks
    if telemetry.gpu_temp_c == 0.0 && telemetry.power_w <= 25.0 {
        return SafetyStatus::Ok;
    }
    if telemetry.gpu_temp_c > 85.0 {
        return SafetyStatus::Critical(format!(
            "GPU thermal: {:.0}°C exceeds 85°C", telemetry.gpu_temp_c
        ));
    }
    if telemetry.power_w > 350.0 {
        return SafetyStatus::Critical(format!(
            "GPU power: {:.0}W exceeds 350W (TDP=360W)", telemetry.power_w
        ));
    }
    SafetyStatus::Ok
}
```

### 4. `src/main.rs` — Wire into main loop
After line 103 (`let telemetry = ...`), add safety check with rate limiting:

```rust
// Safety check every ~1 second (10 steps at 100ms)
if step_count % 10 == 0 {
    match HardwareBridge::check_safety(&telemetry) {
        SafetyStatus::Critical(msg) => {
            eprintln!("[relay] SAFETY CRITICAL: {msg}");
            if let Err(e) = HardwareBridge::apply_emergency_brake(0.5) {
                eprintln!("[relay] Emergency brake failed: {e}");
            }
        }
        SafetyStatus::Warn(msg) => {
            eprintln!("[relay] SAFETY WARN: {msg}");
        }
        SafetyStatus::Ok => {}
    }
}
```
